116:     function generateArc() {
117:         arcContainer.innerHTML = '';
118:         arcItems = [];
119:         circleAngles = [];
120: 
121:         let activeZones = [];
122:         for (let i = 1; i <= 6; i++) {
123:             let en = document.getElementById('cfg_z' + i + '_en');
124:             let val = document.getElementById('cfg_z' + i + '_val');
125:             if (en && en.checked && val) {
126:                 activeZones.push(parseFloat(val.value) || 1.0);
127:             }
128:         }
129:         if (activeZones.length === 0) activeZones = [1.0];
130: 
131:         let isLeftHanded = Boolean(window.appConfig && window.appConfig.leftHandedMode);
132:         const W = window.innerWidth;
133:         const H = window.innerHeight;
134: 
135:         const eqBtn = document.getElementById('btn-eqModal');
136:         if (eqBtn) {
137:             const eqRect = eqBtn.getBoundingClientRect();
138:             currentAnchor.x = eqRect.left + eqRect.width / 2;
139:             currentAnchor.y = eqRect.top + eqRect.height / 2;
140:         } else {
141:             currentAnchor.y = H - 135;
142:             currentAnchor.x = isLeftHanded ? 31 : (W - 31);
143:         }
144: 
145:         // Строим плавную дугу вокруг опорной точки (от середины экрана через край вниз к доку)
146:         // и параметризуем её по длине дуги, чтобы расстояние между кружками было идеально равным!
147:         const startDeg = -108; // Верхний кружок (ниже поиска, правее центра)
148:         const endDeg   = -204; // Нижний кружок (внизу у панели навигации)
149:         const rTop     = Math.min(W * 0.76, H * 0.35);
150:         const rMid     = Math.min(W * 0.78, H * 0.36);
151:         const rBot     = Math.min(W * 0.58, H * 0.26);
152: 
153:         function sampleCurvePoint(u) {
154:             const deg = startDeg + u * (endDeg - startDeg);
155:             const rad = deg * Math.PI / 180;
156:             // Плавный профиль радиуса: широкий изгиб посередине и аккуратный заход вниз
157:             const r = (1 - u) * (1 - u) * rTop + 2 * (1 - u) * u * rMid + u * u * rBot;
158:             const dxRH = r * Math.cos(rad);
159:             const dy = r * Math.sin(rad);
160:             const x = currentAnchor.x + (isLeftHanded ? -dxRH : dxRH);
161:             const y = currentAnchor.y + dy;
162:             return {
163:                 x: Math.max(30, Math.min(W - 30, x)),
164:                 y: Math.max(150, Math.min(H - 32, y))
165:             };
166:         }
167: 
168:         const STEPS = 240;
169:         const samples = [];
170:         let totalLen = 0;
171:         for (let k = 0; k <= STEPS; k++) {
172:             const pt = sampleCurvePoint(k / STEPS);
173:             if (k > 0) {
174:                 totalLen += Math.hypot(pt.x - samples[k - 1].x, pt.y - samples[k - 1].y);
175:             }
176:             samples.push({ x: pt.x, y: pt.y, s: totalLen });
177:         }
178: 
179:         const count = activeZones.length;
180:         const positions = [];
181:         for (let i = 0; i < count; i++) {
182:             const targetS = count === 1 ? (totalLen * 0.5) : (totalLen * (i / (count - 1)));
183:             let best = samples[0];
184:             for (let k = 0; k < samples.length; k++) {
185:                 if (samples[k].s >= targetS) {
186:                     best = samples[k];
187:                     break;
188:                 }
189:             }
190:             positions.push(best);
191:         }
192: 
193:         activeZones.forEach((val, index) => {
194:             if (index >= positions.length) return;
195: 
196:             let el = document.createElement('div');
197:             el.className = 'arc-item';
198:             el.innerText = 'x' + (val % 1 === 0 ? val.toFixed(1) : val.toString());
199:             el.dataset.val = val;
200: 
201:             el.style.left = `${positions[index].x - 24}px`;
202:             el.style.top = `${positions[index].y - 24}px`;
203: 
204:             let angle = Math.atan2(positions[index].y - currentAnchor.y, positions[index].x - currentAnchor.x) * 180 / Math.PI;
205:             circleAngles.push(angle);
206: 
207:             arcContainer.appendChild(el);
208:             arcItems.push(el);
209:         });
210: 
211:         arcContainer.style.display = 'block';
212:         setTimeout(() => { arcContainer.style.opacity = '1'; }, 10);
213: 
214:         drawDebugWheel(currentAnchor.x, currentAnchor.y, circleAngles);
215:     }

280:     function handleTouchMove(e) {
281:         let curX = getX(e);
282:         let curY = getY(e);
283:         let deltaY = curY - startY;
284: 
285:         if (scrubSide === 'center') {
286:             if (Math.abs(deltaY) > 15 && pressTimer) {
287:                 clearTimeout(pressTimer);
288:                 pressTimer = null;
289:             }
290:             return;
291:         }
292: 
293:         let threshold = window.innerHeight * 0.05; 
294: 
295:         if (Math.abs(deltaY) > threshold || isScrubbing) {
296:             isSeekingAllowed = false; 
297:             if (pressTimer) { clearTimeout(pressTimer); pressTimer = null; }
298: 
299:             if (!isScrubbing) {
300:                 isScrubbing = true;
301:                 if (window.scrubState.locked) window.scrubState.locked = false;
302:             }
303: 
304:             if (!isPullingDown) {
305:                 isPullingDown = true;
306:                 overlay.style.display = 'block'; 
307:                 generateArc(); 
308:                 let defSpeed = parseFloat(window.appConfig.defaultScrubSpeed) || 2.0;
309:                 if (typeof window.startScrub === 'function') window.startScrub(scrubSide, defSpeed);
310:             }
311:             
312:             let dx = curX - currentAnchor.x;
313:             let dy = curY - currentAnchor.y;
314:             let fingerAngle = Math.atan2(dy, dx) * 180 / Math.PI;
315: 
316:             let minDiff = Infinity;
317:             let idx = 0;
318:             
319:             circleAngles.forEach((ang, i) => {
320:                 let diff = Math.abs(angleDiffSigned(fingerAngle, ang));
321:                 if (diff < minDiff) {
322:                     minDiff = diff;
323:                     idx = i;
324:                 }
325:             });
326: 
327:             if (!arcItems[idx].classList.contains('active')) {
328:                 arcItems.forEach(el => el.classList.remove('active'));
329:                 arcItems[idx].classList.add('active');
330:                 currentTempSpeed = parseFloat(arcItems[idx].dataset.val);
331:                 if (typeof window.startScrub === 'function') window.startScrub(scrubSide, currentTempSpeed);
332:                 try { navigator.vibrate(15); } catch(err){}
333:             }
334:         }
335:     }
